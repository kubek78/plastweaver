# plastweaver - assembler settings (AssemblerComplete_v16.jl)
#
# plastweaver writes one config per library from this template
# (results/LIBRARY/seed/LIBRARY_config.txt). The following keys are set automatically for
# every library and must not be added here: Project name, Single/Paired, Forward reads,
# Reverse reads, Seed Input, Output path, Genome Range (from the reference chosen by SeedRef;
# 100000-180000 without one), Plastid panel, Reference dataset, Annotation dir,
# Annotation dataset.
#
# Settings to check for your libraries:
#   Read pairs to process   pairs read from the start of the files (0 = all pairs)
#   Read Length             read length of the library
#   Insert size             fragment length of the library (not estimated automatically)
#   Max concurrent assemblies   parallel assemblies per library (memory use grows with it)
# The other values are the tested defaults; change them only with a reason.

Assembly type = chl

Read Length = 150

# First N pairs, keeping pair synchronization. 0 = all pairs.
# For an initial run you can use 200000; insufficient organelle coverage can
# then cause fragmentation. This is not representative random subsampling.
Read pairs to process = 9000000

# UInt64 k-mers support every value 1..32, including T^32.
K-mer = 31
Minimizer window = 10
# GLOBAL fraction of ACTUALLY LOADED reads (R1+R2 in PE), not per worker.
# Keys above either frequency limit are removed entirely; lists are not truncated.
# In an organelle-enriched library, increase the fraction if useful keys vanish.
Max minimizer frequency = 0.01
Max postings per minimizer = 100000
Batch size = 10000

# FASTQ quality encoding: Phred+33. 0 preserves bases regardless of quality;
# 20 masks bases below Q20 to N. N never counts as an alignment match.
Min base quality = 0

# True support from distinct pairs, with conflicting pair votes excluded.
Min overlap = 50
Min coverage = 3
Min identity = 0.98
Consensus fraction = 0.80
Consensus ratio = 2.0
Max extension = 32

Min seed hits = 2
Seed correction = true
Seeds per record = 2
Max initial contigs = 8
Max branch paths = 4
Branch min fraction = 0.20
Max iterations = 20000
Max candidates = 100000
Max concurrent assemblies = 4

# Insert = complete DNA fragment (outer distance), NOT gap between reads.
# There is no automatic estimation; set this from library QC/read mapping.
Insert size = 350
Insert tolerance = 0.30
Paired validation = true
Min paired support = 3

# Overlap merging, without filling unsupported gaps with Ns.
# Both endpoints must have only one supported partner.
Scaffolding = true
Scaffold min overlap = 100
Scaffold max overlap = 5000
Scaffold identity = 0.99
Scaffold rounds = 8

# Separate closure criteria. In PE, both spanning reads and FR pairs are needed.
Circular min overlap = 150
Circular max overlap = 5000
Circular identity = 0.995
Circular min reads = 3
Circular check every = 200
Junction anchor = 20

# Additional FASTA for already circular chloroplasts, rotated to LSC-IR-SSC-IR.
# Detection uses the sequence; this does not assemble missing repeat copies.
Normalize plastome = true
IR min length = 1000
IR min identity = 0.995
IR min SC length = 100

Min contig length = 300
# true also saves all raw assembly paths before deduplication.
Debug mode = false

# Live stage progress, elapsed wall time and current process RSS in decimal GB.
# Old configs without these entries enable the same defaults.
# Terminal: one refreshed line. Redirected output/logs: periodic plain lines.
Show progress = true
Progress interval = 1.0
Progress log interval = 30.0
